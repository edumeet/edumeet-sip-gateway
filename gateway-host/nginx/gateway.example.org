server {
	server_name gateway.example.org;
	server_tokens off;

	location / {
		# HTTPLauncher has no authentication: only the central host may send commands.
		allow 203.0.113.10;
		deny all;
		proxy_pass http://127.0.0.1:8100;
		proxy_set_header Host $host;
	}

	listen 443 ssl;
	ssl_certificate /etc/letsencrypt/live/gateway.example.org/fullchain.pem;
	ssl_certificate_key /etc/letsencrypt/live/gateway.example.org/privkey.pem;
	include /etc/letsencrypt/options-ssl-nginx.conf;
	ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
}

server {
	server_name gateway.example.org;
	server_tokens off;

	listen 80;
	return 404;
}
